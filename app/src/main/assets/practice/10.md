Java-приложение, блокчейн хранилище, реализовать через JMS изменение данных в этом хранилище. Реализацию работы блокчейн писать не надо.

```java
@Stateless
public class BlockchainTransactionProducer {

    @Resource(lookup = "java:/ConnectionFactory")
    private ConnectionFactory connectionFactory;

    @Resource(lookup = "java:/queue/BlockchainQueue")
    private Queue blockchainQueue;

    public void submitTransaction(BlockchainTransaction transaction) {
        try (Connection connection = connectionFactory.createConnection()) {
            Session session = connection.createSession(
                false,
                Session.AUTO_ACKNOWLEDGE
            );

            MessageProducer producer = session.createProducer(blockchainQueue);

            producer.setDeliveryMode(DeliveryMode.PERSISTENT);

            ObjectMessage message = session.createObjectMessage(transaction);
            producer.send(message);
        } catch (JMSException e) {
            throw new RuntimeException("Ошибка отправки транзакции", e);
        }
    }
}

@MessageDriven(
    activationConfig = {
        @ActivationConfigProperty(
            propertyName = "destinationType",
            propertyValue = "jakarta.jms.Queue"
        ),
        @ActivationConfigProperty(
            propertyName = "destination",
            propertyValue = "java:/queue/BlockchainQueue"
        ),
        @ActivationConfigProperty(
            propertyName = "maxSession",
            propertyValue = "1"
        )
    }
)
public class BlockchainTransactionMDB implements MessageListener {

    @Inject
    BlockchainStorage blockchainStorage;

    @Override
    public void onMessage(Message message) {
        try {
            if (message instanceof ObjectMessage objectMessage) {
                BlockchainTransaction transaction =
                    (BlockchainTransaction) objectMessage.getObject();
                blockchainStorage.addTransaction(transaction);
        } catch (JMSException e) {
            // Если упали — сообщение вернётся в очередь
            throw new RuntimeException("Ошибка обработки транзакции", e);
        }
    }
}
```
