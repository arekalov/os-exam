Планировщик на Quartz, осуществляющий проверку на то, не болен ли студент COVID-19, и не опасен ли он для окружающих. Планировщик должен работать следующим образом: - Каждый день в 10:00 и 22:00 измерять температуру каждому студенту (функция checkTemperature). - Если температура выше, чем 37,2°С в течение двух, или более наблюдений подряд, вызвать санитаров с ПЦР-тестом (функция callParamedic). - Если результаты теста положительные (функция checkTestResults), заблокировать пропуска в здание университета для указанного студента, всех его одногруппников, и преподавателей, контактировавших с ним в последние 10 дней.

```java
public class CheckCovidJob implements Job {
    @Override
    public void execute(JobExecutionContext context) throws JobExecutionException {
        JobDataMap dataMap = context.getJobDetail().getJobDataMap();
        IsuService isuService = (IsuService) dataMap.get("isuService");
        MedicalService medService = (MedicalService) dataMap.get("medService");


        List<Student> students = isuService.getAllStudents();
        for (Student student: students) {
            float temperature = medService.checkTemperature(student);
            isuService.writeTemperature(student, temperature);
            if (isuService.hasBadTemperatureMoreThanTwoDays(student)) {
                Paramedic paramedic = medService.callParamedic();
                if (paramedic.checkTestResults(student).isNotOk()) {
                    isuService.blockNecceceryPeopleByStudent(student);
                }
            }
        }
    }
}

public class CovidCheckScheduler {
    IsuService isuService;
    MedicalService medService;
    //constructor должен быть здесь

    public void start() {
         private Scheduler scheduler = StdSchedulerFactory.getDefaultScheduler();

         JobDataMap dataMap = new JobDataMap();
         dataMap.put("isuService", isuService);
         dataMap.put("medService", medService);


         JobDetail job = JobBuilder.newJob(CheckCovidJob.class)
                .withIdentity("checkCovidJob", "common")
                .usingJobData(dataMap)
                .build();

        Trigger trigger = TriggerBuilder.newTrigger()
            .withIdentity("checkCovidTrigger", "common")
            .WithCronSchedule("0 0 10,22 * * *")
            .build();

        scheduler.scheduleJob(job, trigger);
        scheduler.start();
    }

    public static void main(String[] args) {
        CovidCheckScheduler().start();

        Thread.sleep(Long.MAX_VALUE);
    }
}
```
