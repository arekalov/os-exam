Написать Spring щедулер для робоЦопы: рабочий день с 13 до 17, перерыв 13-15, работает по будням, начинает каждые 5 минут. Если два нарушены, то вызвать printLetter(student): List<Student> checkOutwear(), List<Chair> checkAnglePosition() больше 5, List<Student> checkFood(), List<Student> checkDrinks()


```java
@Service 
public class SchedulerService {
    @Autowired
    LetterService letterService;

    @Autowired
    ChecksService checksService;

    @Autowired
    StudentsService studentsService;

    // важно: первая колонка - секунду обязательна
    @Scheduled(cron = "* */5 11-12,15-16 * * 1-5")
    void tryToSendLetter() {
        List<Students> students = studentsService.getStudents();
        
        List<Students> incorrectWearStudents = checksService.checkOutWear();
        List<Students> incorrectChairStudents = checksService.checkChairAngles(5);
        List<Students> hasFoodStudents = checksService.checkFood();
        List<Students> hasDrinksStudents = checksService.checkDrinks();

        for (Student student: students) {
            int counter = 0;
            if (incorrectWearStudents.contains(student)) {
                counter++;
            }
            if (incorrectChairStudents.contains(student)) {
                counter++;
            }
            if (hasDrinksStudents.contains(student)) {
                counter++;
            }
            if (hasFoodStudents.contains(student)) {
                counter++;
            }

            if (counter >= 2) {
                letterService.sendLetter(student);
            }
        }
    }
}
```
