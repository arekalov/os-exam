Написать класс который с помощью JMS делает рассылку научных статей случайно выбранным рецензентам (выбирает их из пула)

```java
@Stateless
public class ArticleSubmissionService {

    @Resource(lookup = "java:/ConnectionFactory")
    private ConnectionFactory connectionFactory;

    @Resource(lookup = "java:/queue/ReviewerQueue")
    private Queue reviewerQueue;

    @Inject
    ReviewerSelectionService reviewerSelectionService;

    public void submitArticleForReview(ScientificArticle article) {
        List<Reviewer> selectedReviewers = 
            reviewerSelectionService.selectReviewers(article);

        try (Connection connection = connectionFactory.createConnection()) {
            Session session = connection.createSession(
                false,
                Session.AUTO_ACKNOWLEDGE
            );

            MessageProducer producer = session.createProducer(reviewerQueue);

            for (Reviewer reviewer : selectedReviewers) {
                ObjectMessage message = session.createObjectMessage(article);
                producer.send(message);
            }

        } catch (JMSException e) {
            throw new RuntimeException("Ошибка отправки", e);
        }
    }
}

@MessageDriven(
    activationConfig = {
        @ActivationConfigProperty(
            propertyName = "destinationType",
            propertyValue = "jakarta.jms.Queue"
        ),
        @ActivationConfigProperty(
            propertyName = "destination",
            propertyValue = "java:/queue/ReviewerQueue"
        )
    }
)
public class ReviewerNotificationMDB implements MessageListener {
    @Inject
    NotificationService notificationService;

    @Override
    public void onMessage(Message message) {
        try {
            if (message instanceof ObjectMessage objectMessage) {
                ReviewNotification notification =
                    (ReviewNotification) objectMessage.getObject();

                notificationService.sendEmail(notification);
            }
        } catch (JMSException e) {
            throw new RuntimeException("Ошибка обработки сообщения", e);
        }
    }
}
```
