Планировщик на Jakarta EE Timer Services, блокирующий стулом вход в аудиторию через 5 минут после начала лекционного занятия. Блокирование при следующих условиях: - Пара началась 5 мин назад (необходимо использовать расписание пар ИТМО); - В данной аудитории в данный момент проходит лекция (можно запросить через ИСУ API); - Лектор зашел в аудиторию и включил проектор.

```java
@Singleton
public class DoorBlockerScheduler {
    public static final int TSOPA_ISU_ID = 123456;
    public static final int AUDITORY_ID = 2112;

    @Inject
    IsuCommonService isuService;
   
   @Schedules({
        @Schedule(cron = "0 15 8 * * 1-5"),
        @Schedule(cron = "0 55 9 * * 1-5"),
        @Schedule(cron = "0 35 11 * * 1-5"),
        // Дальше не помню расписание но нужно добавлять +5 минут
   })
   public void tryToLockDoor() {
        LocalDateTime currentDate = LocalDateTime.now();
        boolean hasLectionInThisTime = isuService.hasLectionByDateTime(currentDate, AUDITORY_ID);
        boolean hasLectorInCampus = isuService.checkLectorInCampus(TSOPA_ISU_ID);
        boolean hasProjectorSwitchedOn = isuService.isProjectorSwitchedOn(AUDITORY_ID);

        if (hasLectionInThisTime && hasLectorInCampus && hasProjectorSwitchedOn) {
            isuService.lockDoorByChair();
        }
   }
}
```
